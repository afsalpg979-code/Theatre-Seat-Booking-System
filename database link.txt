CREATE TABLE bookings (
    id INT(11) AUTO_INCREMENT PRIMARY KEY,
    name VARCHAR(100),
    phone VARCHAR(20),
    seats TEXT,
    movie VARCHAR(100),
    totalAmount DECIMAL(10, 2),
    time DATETIME DEFAULT CURRENT_TIMESTAMP
);


CREATE TABLE admin (
    id INT(11) AUTO_INCREMENT PRIMARY KEY,
    username VARCHAR(50),
    password VARCHAR(255)
);
